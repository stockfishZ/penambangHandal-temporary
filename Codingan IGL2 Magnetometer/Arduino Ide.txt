#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>
#include <TinyGPSPlus.h>
#include <math.h>

/*
  ==========================================================
  ESP32 + OLED SSD1306 + GPS NEO-6M + HMC5883L / QMC5883L
  ==========================================================
  Fitur:
  - Tanpa Hall Linear Sensor
  - 10 menu OLED
  - Rotary encoder untuk pindah menu
  - GPS NEO-6M
  - Magnetometer HMC5883L atau QMC5883L
  - Serial Monitor cocok untuk GUI Tkinter dan CSV QGIS

  Wiring:
  OLED SDA / HMC SDA  -> GPIO18
  OLED SCL / HMC SCL  -> GPIO19
  GPS TX              -> GPIO16
  GPS RX              -> GPIO17
  Encoder CLK         -> GPIO32
  Encoder DT          -> GPIO33
  Encoder SW          -> GPIO25
  VCC                 -> 3V3 / 5V sesuai modul
  GND                 -> GND
*/

// ================= OLED =================
#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
#define OLED_RESET -1
#define OLED_ADDRESS 0x3C

#define I2C_SDA 18
#define I2C_SCL 19

Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, OLED_RESET);

// ================= GPS =================
TinyGPSPlus gps;
HardwareSerial gpsSerial(2);

#define GPS_RX 16
#define GPS_TX 17

// Jika GPS tidak terbaca, coba ganti 115200 menjadi 9600
#define GPS_BAUD 115200

// ================= ROTARY ENCODER =================
#define ENC_CLK 32
#define ENC_DT  33
#define ENC_SW  25

int menuIndex = 0;
const int totalMenu = 10;

int lastCLK = HIGH;
int lastSW = HIGH;

unsigned long lastRotateTime = 0;
unsigned long lastButtonTime = 0;

// ================= MAGNETOMETER =================
#define HMC_ADDR 0x1E
#define QMC_ADDR 0x0D

int magType = 0;
// 0 = tidak terdeteksi
// 1 = HMC5883L
// 2 = QMC5883L

int16_t magX = 0;
int16_t magY = 0;
int16_t magZ = 0;

float magTotal = 0.0;
float headingDeg = 0.0;
String arahKompas = "-";
bool magOK = false;

// ================= TIMER =================
unsigned long lastOLED = 0;
unsigned long lastSerial = 0;

// ================= I2C HELPER =================
bool i2cDeviceExists(byte address) {
  Wire.beginTransmission(address);
  return Wire.endTransmission() == 0;
}

void writeRegister(byte address, byte reg, byte value) {
  Wire.beginTransmission(address);
  Wire.write(reg);
  Wire.write(value);
  Wire.endTransmission();
}

// ================= TEXT HELPER =================
String getMagTypeText() {
  if (magType == 1) return "HMC5883L";
  if (magType == 2) return "QMC5883L";
  return "NOT FOUND";
}

String getGPSValidText() {
  if (gps.location.isValid()) return "VALID";
  return "NO FIX";
}

String getDirection(float deg) {
  if (deg >= 337.5 || deg < 22.5) return "UTARA";
  else if (deg < 67.5) return "TIMUR LAUT";
  else if (deg < 112.5) return "TIMUR";
  else if (deg < 157.5) return "TENGGARA";
  else if (deg < 202.5) return "SELATAN";
  else if (deg < 247.5) return "BARAT DAYA";
  else if (deg < 292.5) return "BARAT";
  else return "BARAT LAUT";
}

String getMagInterpretasi(float nilai) {
  // Batas interpretasi sederhana.
  // Boleh disesuaikan setelah kalibrasi lapangan.
  if (nilai < 300) return "RENDAH";
  else if (nilai <= 600) return "NORMAL";
  else return "TINGGI";
}

// ================= INIT MAGNETOMETER =================
void initMagnetometer() {
  if (i2cDeviceExists(HMC_ADDR)) {
    magType = 1;

    // Konfigurasi HMC5883L
    writeRegister(HMC_ADDR, 0x00, 0x70);
    writeRegister(HMC_ADDR, 0x01, 0x20);
    writeRegister(HMC_ADDR, 0x02, 0x00);

    Serial.println("Magnetometer: HMC5883L");
  }
  else if (i2cDeviceExists(QMC_ADDR)) {
    magType = 2;

    // Konfigurasi QMC5883L
    writeRegister(QMC_ADDR, 0x0B, 0x01);
    writeRegister(QMC_ADDR, 0x09, 0x1D);

    Serial.println("Magnetometer: QMC5883L");
  }
  else {
    magType = 0;
    Serial.println("Magnetometer: NOT FOUND");
  }
}

// ================= READ MAGNETOMETER =================
bool readMagnetometer() {
  if (magType == 1) {
    // HMC5883L urutan data: X, Z, Y
    Wire.beginTransmission(HMC_ADDR);
    Wire.write(0x03);

    if (Wire.endTransmission() != 0) {
      magOK = false;
      return false;
    }

    Wire.requestFrom(HMC_ADDR, 6);

    if (Wire.available() < 6) {
      magOK = false;
      return false;
    }

    magX = (int16_t)((Wire.read() << 8) | Wire.read());
    magZ = (int16_t)((Wire.read() << 8) | Wire.read());
    magY = (int16_t)((Wire.read() << 8) | Wire.read());
  }
  else if (magType == 2) {
    // QMC5883L urutan data: X, Y, Z little endian
    Wire.beginTransmission(QMC_ADDR);
    Wire.write(0x00);

    if (Wire.endTransmission() != 0) {
      magOK = false;
      return false;
    }

    Wire.requestFrom(QMC_ADDR, 6);

    if (Wire.available() < 6) {
      magOK = false;
      return false;
    }

    byte xl = Wire.read();
    byte xh = Wire.read();
    byte yl = Wire.read();
    byte yh = Wire.read();
    byte zl = Wire.read();
    byte zh = Wire.read();

    magX = (int16_t)((xh << 8) | xl);
    magY = (int16_t)((yh << 8) | yl);
    magZ = (int16_t)((zh << 8) | zl);
  }
  else {
    magOK = false;
    return false;
  }

  magTotal = sqrt((float)magX * magX + (float)magY * magY + (float)magZ * magZ);

  headingDeg = atan2((float)magY, (float)magX) * 180.0 / PI;

  if (headingDeg < 0) {
    headingDeg += 360.0;
  }

  arahKompas = getDirection(headingDeg);
  magOK = true;

  return true;
}

// ================= ENCODER =================
void nextMenu() {
  menuIndex++;
  if (menuIndex >= totalMenu) menuIndex = 0;

  Serial.print("Menu: ");
  Serial.println(menuIndex + 1);

  lastOLED = 0;
}

void previousMenu() {
  menuIndex--;
  if (menuIndex < 0) menuIndex = totalMenu - 1;

  Serial.print("Menu: ");
  Serial.println(menuIndex + 1);

  lastOLED = 0;
}

void updateEncoder() {
  int clkState = digitalRead(ENC_CLK);

  if (clkState != lastCLK) {
    if (millis() - lastRotateTime > 60) {
      lastRotateTime = millis();

      if (clkState == LOW) {
        int dtState = digitalRead(ENC_DT);

        if (dtState == HIGH) {
          nextMenu();
        } else {
          previousMenu();
        }
      }
    }

    lastCLK = clkState;
  }

  int swState = digitalRead(ENC_SW);

  if (swState != lastSW && millis() - lastButtonTime > 300) {
    lastButtonTime = millis();

    if (lastSW == HIGH && swState == LOW) {
      menuIndex = 0;
      Serial.println("Button: HOME");
      lastOLED = 0;
    }

    lastSW = swState;
  }
}

// ================= OLED HELPER =================
void oledHeader(const char* title) {
  display.clearDisplay();
  display.setTextColor(SSD1306_WHITE);
  display.setTextSize(1);
  display.setCursor(0, 0);
  display.println(title);
  display.drawLine(0, 10, 127, 10, SSD1306_WHITE);
}

  display.display();
}

// ================= SETUP =================
void setup() {
  Serial.begin(115200);
  delay(1000);

  Wire.begin(I2C_SDA, I2C_SCL);

  if (!display.begin(SSD1306_SWITCHCAPVCC, OLED_ADDRESS)) {
    Serial.println("OLED gagal dimulai!");
    while (true);
  }

  display.clearDisplay();
  display.setTextColor(SSD1306_WHITE);
  display.setTextSize(1);
  display.setCursor(0, 0);
  display.println("Starting system...");
  display.setCursor(0, 16);
  display.println("OLED OK");
  display.setCursor(0, 32);
  display.println("GPS + MAG");
  display.setCursor(0, 48);
  display.println("10 Menu Ready");
  display.display();

  gpsSerial.begin(GPS_BAUD, SERIAL_8N1, GPS_RX, GPS_TX);

  pinMode(ENC_CLK, INPUT_PULLUP);
  pinMode(ENC_DT, INPUT_PULLUP);
  pinMode(ENC_SW, INPUT_PULLUP);

  lastCLK = digitalRead(ENC_CLK);
  lastSW = digitalRead(ENC_SW);

  initMagnetometer();

  delay(1500);
  updateOLED();

  Serial.println();
  Serial.println("ESP32 GPS + Magnetometer siap");
  Serial.println("Tanpa Hall Sensor Linear");
  Serial.println("10 Menu OLED aktif");
  Serial.print("GPS Baud: ");
  Serial.println(GPS_BAUD);
  Serial.println();
}

// ================= LOOP =================
void loop() {
  updateEncoder();

  while (gpsSerial.available() > 0) {
    gps.encode(gpsSerial.read());
  }

  readMagnetometer();

  if (millis() - lastOLED >= 500) {
    lastOLED = millis();
    updateOLED();
  }

  if (millis() - lastSerial >= 1000) {
    lastSerial = millis();
    printSerialData();
  }
}
