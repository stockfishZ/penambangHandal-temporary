    # ======================================================
    # SERIAL

    def get_ports(self):
        if list_ports is None:
            return ["COM3"]
        ports = [p.device for p in list_ports.comports()]
        return ports if ports else ["COM3"]

    def refresh_ports(self):
        self.port_combo["values"] = self.get_ports()

    def connect(self):
        if serial is None:
            messagebox.showerror("pyserial belum ada", "Install dulu:\n\npip install pyserial")
            return

        self.disconnect(close_status=False)

        try:
            port = self.port_var.get().strip()
            baud = int(self.baud_var.get().strip())

            self.ser = serial.Serial(port, baud, timeout=0.02)
            time.sleep(1.2)
            self.ser.reset_input_buffer()
            self.connected = True

            self.status_label.config(
                text=f"Serial: CONNECTED ke {port} @ {baud}.",
                fg="#15803d"
            )
        except Exception as e:
            self.ser = None
            self.connected = False
            self.status_label.config(text=f"Serial gagal connect: {e}", fg="#b91c1c")

    def disconnect(self, close_status=True):
        try:
            if self.ser is not None and self.ser.is_open:
                self.ser.close()
        except Exception:
            pass

        self.ser = None
        self.connected = False

        if close_status:
            self.status_label.config(text="Serial: disconnected.", fg="#b45309")

    def serial_loop(self):
        if self.connected and self.ser is not None:
            try:
                count = 0
                while self.ser.in_waiting and count < 150:
                    line = self.ser.readline().decode("utf-8", errors="ignore").strip()
                    count += 1

                    if line:
                        self.line_count += 1
                        self.last_line = line
                        self.parse_line(line)

                if count > 0:
                    self.raw_label.config(text=f"Line #{self.line_count}\n{self.last_line[:55]}")
                    self.update_after_data()

            except Exception as e:
                self.status_label.config(text=f"Serial error: {e}", fg="#b91c1c")
                self.disconnect(close_status=False)

        self.root.after(30, self.serial_loop)

    # ======================================================
    # PARSER
    # ======================================================
    def norm_key(self, text):
        text = text.strip().lower().replace("_", " ")
        return re.sub(r"\s+", " ", text)

    def to_float(self, text):
        m = re.search(r"[-+]?\d+(?:\.\d+)?", text)
        if not m:
            raise ValueError("No number")
        return float(m.group(0))

    def parse_line(self, line):
        if not line or "=====" in line:
            return
        if ":" not in line:
            return

        key, value = line.split(":", 1)
        key = self.norm_key(key)
        value = value.strip()

        try:
            if key == "menu":
                self.menu = int(round(self.to_float(value)))

            elif key == "gps valid":
                self.gps_valid = value.lower() in ["true", "valid", "ok", "1"]

            elif key in ["gps sat", "satellites", "satelit"]:
                self.sat = int(round(self.to_float(value)))
                self.g_sat.append(float(self.sat))

            elif key in ["lat", "latitude", "gps latitude"]:
                if "belum" in value.lower():
                    self.gps_valid = False
                else:
                    self.lat = self.to_float(value)
                    self.gps_valid = True

            elif key in ["lng", "lon", "longitude", "gps longitude"]:
                if "belum" in value.lower():
                    self.gps_valid = False
                else:
                    self.lng = self.to_float(value)
                    self.gps_valid = True
                    self.add_track_point()

            elif key in ["altitude", "alt"]:
                if "belum" in value.lower() or value == "-":
                    self.alt_valid = False
                else:
                    self.altitude = self.to_float(value)
                    self.alt_valid = True

            elif key in ["speed", "spd"]:
                if "belum" in value.lower() or value == "-":
                    self.speed_valid = False
                else:
                    self.speed = self.to_float(value)
                    self.speed_valid = True
                    self.g_speed.append(float(self.speed))

            elif key == "magnetometer":
                self.mag_type = value

            elif key in ["mag x", "magx"]:
                self.mag_x = self.to_float(value)
                self.g_x.append(float(self.mag_x))

            elif key in ["mag y", "magy"]:
                self.mag_y = self.to_float(value)
                self.g_y.append(float(self.mag_y))

            elif key in ["mag z", "magz"]:
                self.mag_z = self.to_float(value)
                self.g_z.append(float(self.mag_z))

            elif key in ["mag total", "b total"]:
                self.mag_total = self.to_float(value)
                self.g_mag.append(float(self.mag_total))

            elif key in ["heading", "heading deg"]:
                self.heading = self.to_float(value)
                self.g_heading.append(float(self.heading))

            elif key in ["direction", "arah"]:
                self.direction = value

            self.last_update = time.strftime("%H:%M:%S")

        except Exception:
            pass

    def mag_category(self):
        if self.mag_total < 300:
            return "RENDAH"
        if self.mag_total <= 600:
            return "NORMAL"
        return "TINGGI"

    def ai_interpretation(self):
        if not self.gps_valid:
            return "GPS belum valid. Data belum layak dianalisis. Tunggu koordinat fix."

        if self.mag_type == "NOT FOUND":
            return "Magnetometer belum terbaca. Cek VCC, GND, SDA, SCL, dan alamat I2C."

        recent = [v for v in list(self.g_mag)[-30:] if v != 0.0]

        if len(recent) < 8:
            if self.sat < 4:
                return "Data masih sedikit dan GPS lemah. Lanjutkan survei di area terbuka."
            if self.mag_total > 600:
                return "Data awal tinggi. Jangan langsung simpulkan anomali; cek jarak dari besi/kabel/HP."
            return "Data awal masih terbatas. Lanjutkan lintasan agar pola anomali lebih terbaca."

        avg = sum(recent) / len(recent)
        mn = min(recent)
        mx = max(recent)
        rng = mx - mn
        diff = self.mag_total - avg

        last5 = recent[-5:]
        trend = last5[-1] - last5[0] if len(last5) >= 2 else 0

        if self.sat < 4:
            gps_note = "GPS lemah; posisi titik kurang dipercaya."
        elif self.sat < 8:
            gps_note = "GPS cukup; interpretasi masih perlu validasi."
        else:
            gps_note = "GPS baik; posisi relatif layak dipakai."

        if self.mag_total > 650 or diff > 80:
            status = "ANOMALI TINGGI"
            sebab = "indikasi objek logam, kabel, pagar, kendaraan, atau batuan bermagnet."
            saran = "Ulangi titik ini 2-3 kali dan jauhkan sensor dari benda logam."
        elif self.mag_total < 250 or diff < -80:
            status = "ANOMALI RENDAH"
            sebab = "indikasi perubahan material bawah permukaan atau efek orientasi sensor."
            saran = "Cek ulang heading sensor dan bandingkan dengan titik sekitar."
        elif rng > 120:
            status = "ZONA TIDAK STABIL"
            sebab = "nilai magnetik berubah besar sepanjang lintasan."
            saran = "Perlu peta kontur/interpolasi di QGIS sebelum disimpulkan."
        elif abs(diff) <= 40 and rng <= 80:
            status = "STABIL"
            sebab = "variasi magnetik kecil terhadap titik sekitar."
            saran = "Belum ada indikasi anomali kuat pada titik ini."
        else:
            status = "TRANSISI"
            sebab = "nilai mulai berubah dari pola sekitar."
            saran = "Lanjutkan lintasan rapat untuk melihat batas anomalinya."

        if trend > 50:
            arah_pola = "Tren naik cepat."
        elif trend < -50:
            arah_pola = "Tren turun cepat."
        else:
            arah_pola = "Tren perubahan pelan."

        return (
            f"Status anomali : {status}\n"
            f"Makna data     : {sebab}\n"
            f"Pola lintasan  : {arah_pola} Nilai sekarang {self.mag_total:.1f}, "
            f"rata-rata {avg:.1f}, selisih {diff:+.1f}, rentang {rng:.1f}.\n"
            f"Kualitas GPS   : {gps_note}\n"
            f"Pola track     : {self.local_track_note()}\n"
            f"Kesimpulan     : {saran}"
        )

    def local_track_note(self):
        stats = self.track_analysis()

        n = stats["jumlah_titik_track"]
        if n < 2:
            return "Track belum cukup; butuh minimal dua titik GPS untuk membaca pola lintasan."

        if stats["jarak_total_xy_m"] < 1.0:
            return "Pergerakan sangat kecil; pembacaan magnetik lebih cocok dianggap titik diam."

        return (
            f"{stats['pola_lintasan']}; jarak total {stats['jarak_total_xy_m']:.2f} m, "
            f"sebaran X {stats['rentang_x_m']:.2f} m, Y {stats['rentang_y_m']:.2f} m, "
            f"Z {stats['rentang_z_m']:.2f} m."
        )

    def track_analysis(self):
        # Analisis GPS Track Preview X/Y/Z.
        # Nilai X/Y/Z dipakai untuk melihat apakah lintasan sudah bergerak,
        # menyebar, lurus, bolak-balik, atau masih terlalu pendek.
        points = list(self.track)

        if not points:
            return {
                "jumlah_titik_track": 0,
                "x_sekarang_m": round(self.local_x, 2),
                "y_sekarang_m": round(self.local_y, 2),
                "z_sekarang_m": round(self.local_z, 2),
                "min_x_m": 0.0,
                "max_x_m": 0.0,
                "rentang_x_m": 0.0,
                "min_y_m": 0.0,
                "max_y_m": 0.0,
                "rentang_y_m": 0.0,
                "min_z_m": 0.0,
                "max_z_m": 0.0,
                "rentang_z_m": 0.0,
                "jarak_total_xy_m": 0.0,
                "jarak_total_xyz_m": 0.0,
                "jarak_dari_awal_xy_m": 0.0,
                "jarak_dari_origin_xy_m": math.sqrt(self.local_x ** 2 + self.local_y ** 2),
                "jarak_step_terakhir_xy_m": 0.0,
                "arah_perpindahan": "belum cukup data",
                "pola_lintasan": "track belum cukup",
                "catatan_z": "Z belum cukup dianalisis",
                "kualitas_track": "rendah"
            }

        xs = [p[0] for p in points]
        ys = [p[1] for p in points]
        zs = [p[2] for p in points]

        min_x, max_x = min(xs), max(xs)
        min_y, max_y = min(ys), max(ys)
        min_z, max_z = min(zs), max(zs)

        range_x = max_x - min_x
        range_y = max_y - min_y
        range_z = max_z - min_z

        total_xy = 0.0
        total_xyz = 0.0
        last_step_xy = 0.0

        for i in range(1, len(points)):
            dx = points[i][0] - points[i - 1][0]
            dy = points[i][1] - points[i - 1][1]
            dz = points[i][2] - points[i - 1][2]

            step_xy = math.sqrt(dx ** 2 + dy ** 2)
            step_xyz = math.sqrt(dx ** 2 + dy ** 2 + dz ** 2)

            total_xy += step_xy
            total_xyz += step_xyz
            last_step_xy = step_xy

        first = points[0]
        last = points[-1]
        dx_total = last[0] - first[0]
        dy_total = last[1] - first[1]
        displacement_xy = math.sqrt(dx_total ** 2 + dy_total ** 2)
        distance_from_origin_xy = math.sqrt(last[0] ** 2 + last[1] ** 2)

        if abs(dx_total) < 0.5 and abs(dy_total) < 0.5:
            arah = "perpindahan sangat kecil"
        elif abs(dx_total) > abs(dy_total) * 1.8:
            arah = "dominan timur/barat"
        elif abs(dy_total) > abs(dx_total) * 1.8:
            arah = "dominan utara/selatan"
        else:
            arah = "diagonal atau campuran"

        if len(points) < 3:
            pola = "track masih sangat pendek"
            kualitas_track = "rendah"
        elif total_xy < 1.0:
            pola = "posisi hampir diam"
            kualitas_track = "rendah"
        elif range_x < 2.0 and range_y < 2.0:
            pola = "sebaran track terlalu sempit"
            kualitas_track = "rendah"
        elif total_xy > 0 and displacement_xy / total_xy > 0.75:
            pola = "lintasan relatif lurus"
            kualitas_track = "sedang sampai baik"
        elif total_xy > 0 and displacement_xy / total_xy < 0.35:
            pola = "lintasan bolak-balik atau berputar"
            kualitas_track = "sedang"
        else:
            pola = "lintasan menyebar cukup baik"
            kualitas_track = "sedang sampai baik"

        if range_z > 8:
            catatan_z = "Z berubah besar; data tinggi GPS perlu divalidasi karena GPS vertikal sering kurang stabil"
        elif range_z > 3:
            catatan_z = "Z berubah sedang; cek apakah perubahan elevasi benar atau noise GPS"
        else:
            catatan_z = "Z relatif stabil"

        return {
            "jumlah_titik_track": len(points),
            "x_sekarang_m": round(self.local_x, 2),
            "y_sekarang_m": round(self.local_y, 2),
            "z_sekarang_m": round(self.local_z, 2),
            "min_x_m": round(min_x, 2),
            "max_x_m": round(max_x, 2),
            "rentang_x_m": round(range_x, 2),
            "min_y_m": round(min_y, 2),
            "max_y_m": round(max_y, 2),
            "rentang_y_m": round(range_y, 2),
            "min_z_m": round(min_z, 2),
            "max_z_m": round(max_z, 2),
            "rentang_z_m": round(range_z, 2),
            "jarak_total_xy_m": round(total_xy, 2),
            "jarak_total_xyz_m": round(total_xyz, 2),
            "jarak_dari_awal_xy_m": round(displacement_xy, 2),
            "jarak_dari_origin_xy_m": round(distance_from_origin_xy, 2),
            "jarak_step_terakhir_xy_m": round(last_step_xy, 2),
            "arah_perpindahan": arah,
            "pola_lintasan": pola,
            "catatan_z": catatan_z,

        }

    def clean_ai_text(self, text, payload=None):
        if not text:
            return ""

        fallback_lines = self.complex_local_interpretation(payload)

        # Qwen kadang menambahkan markdown seperti ###, **, bullet, nomor,
        # atau membuat isi terlalu pendek. Bagian ini membersihkan dan
        # melengkapi hasil agar selalu utuh di Google Sheets.
        text = text.replace("###", "").replace("##", "").replace("#", "")
        text = text.replace("**", "").replace("*", "")
        text = text.replace("：", ":")

        cleaned_lines = []
        for line in text.splitlines():
            line = line.strip()
            line = re.sub(r"^[\-•]+\s*", "", line)
            line = re.sub(r"^\d+[\.)]\s*", "", line)
            line = re.sub(r"\s+", " ", line).strip()
            if line:
                cleaned_lines.append(line)

        important_labels = [
            "Status kritis:",
            "Kualitas data:",
            "Dasar magnetik:",
            "Analisis track X/Y/Z:",
            "Pola spasial:",
            "Potensi bias:",
            "Kemungkinan penyebab:",
            "Tingkat keyakinan:",
            "Tindakan lapangan:"
        ]

        aliases = {
            "Status kritis:": ["status kritis:", "status:", "status anomali:"],
            "Kualitas data:": ["kualitas data:", "kualitas gps:", "kelayakan data:"],
            "Dasar magnetik:": ["dasar magnetik:", "dasar data:", "data magnetik:", "dasar pembacaan:"],
            "Analisis track X/Y/Z:": ["analisis track x/y/z:", "analisis track:", "track x/y/z:", "analisis lintasan:"],
            "Pola spasial:": ["pola spasial:", "pola lintasan:", "hubungan spasial:"],
            "Potensi bias:": ["potensi bias:", "bias:", "potensi gangguan:"],
            "Kemungkinan penyebab:": ["kemungkinan penyebab:", "penyebab:", "dugaan penyebab:"],
            "Tingkat keyakinan:": ["tingkat keyakinan:", "keyakinan:", "confidence:"],
            "Tindakan lapangan:": ["tindakan lapangan:", "tindakan:", "saran:", "rekomendasi:"]
        }

        def is_line_complete(line, label):
            if not line or ":" not in line:
                return False

            content = line.split(":", 1)[1].strip()
            content_lower = content.lower().strip()

            bad_words = ["belum lengkap", "[", "]", "isi ", "contoh", "...", "n/a"]
            if any(bad in content_lower for bad in bad_words):
                return False

            # Minimal 9 kata setelah label agar tidak muncul jawaban terlalu pendek
            # seperti "stabil", "GPS lemah", atau kalimat yang belum matang.
            if len(content.split()) < 9:
                return False

            # Deteksi jawaban yang berhenti di tengah kata atau di tengah ide.
            # Contoh masalah nyata: "belum cukup untuk menentukan pola spasi".
            incomplete_fragments = [
                "pola spasi", "menentukan pola spasi", "hubungan spasi",
                "dan", "atau", "karena", "dengan", "untuk", "yang",
                "di", "ke", "dari", "pada", "dalam", "sebagai",
                "namun", "tetapi", "sehingga", "agar", "terhadap",
                "berdasarkan", "misalnya", "seperti"
            ]

            if any(content_lower.endswith(fragment) for fragment in incomplete_fragments):
                return False

            if content_lower[-1:] in [",", ";", ":", "-", "/"]:
                return False

            if content.count("(") > content.count(")"):
                return False

            # Bagian Pola spasial harus benar-benar menyebut hubungan X/Y/Z
            # atau menjelaskan bahwa pola belum cukup, bukan berhenti di kata terpotong.
            if label == "Pola spasial:" and "spasi" in content_lower and "spasial" not in content_lower:
                return False

            return True

        selected = []
        lower_lines = [line.lower() for line in cleaned_lines]

        for label in important_labels:
            found_line = None
            for idx, low in enumerate(lower_lines):
                if any(low.startswith(alias) for alias in aliases[label]):
                    content = cleaned_lines[idx].split(":", 1)
                    if len(content) == 2:
                        candidate = f"{label} {content[1].strip()}"
                    else:
                        candidate = cleaned_lines[idx]
                    found_line = candidate
                    break

            # Kalau model memberi jawaban terlalu pendek atau terpotong,
            # gunakan interpretasi lokal yang lengkap sesuai data sensor.
            if is_line_complete(found_line, label):
                selected.append(found_line)
            else:
                selected.append(fallback_lines[label])

        return "\n".join(selected)


    def ollama_interpretation(self, payload=None):
        fallback = self.ai_interpretation()

        if requests is None:
            return fallback

        if not USE_OLLAMA_AI:
            return fallback

        if not self.ready_to_log():
            return fallback

        try:
            prompt = self.build_ollama_prompt(payload)
            response = requests.post(
                OLLAMA_URL,
                json={
                    "model": OLLAMA_MODEL,
                    "prompt": prompt,
                    "stream": False,
                    "options": {
                        "temperature": 0.25,
                        "top_p": 0.9,
                        "repeat_penalty": 1.18,
                        "num_predict": 560
                    },
                    "keep_alive": "10m"
                },
                timeout=OLLAMA_TIMEOUT
            )

            if response.status_code != 200:
                return fallback + f"\n\nCatatan Ollama: HTTP {response.status_code}."

            result = response.json()
            text = result.get("response", "").strip()

            if not text:
                return fallback

            cleaned_text = self.clean_ai_text(text, payload)
            if not cleaned_text:
                return fallback

            return cleaned_text

        except Exception as e:
            return fallback + f"\n\nCatatan Ollama: belum terbaca ({e})."

    def auto_send_to_sheet(self):
        if not self.sheet_upload:
            return

        if not self.ready_to_log():
            self.sheet_status = "Menunggu GPS & magnetometer valid"
            self.refresh_ui()
            return

        if self.sheet_busy:
            self.sheet_status = "Menunggu interpretasi sebelumnya selesai"
            return

        now = time.time()
        if now - self.last_sheet_time < getattr(self, "log_interval", 1.0):
            return

        payload = self.make_payload()
        self.last_sheet_time = now
        self.sheet_busy = True

        threading.Thread(
            target=self.send_sheet_thread,
            args=(payload,),
            daemon=True
        ).start()

    def send_sheet_thread(self, payload):
        try:
            if USE_OLLAMA_AI:
                self.sheet_status = "Menganalisis 9 aspek sensor + track X/Y/Z dengan Ollama..."
                self.root.after(0, self.refresh_ui)
                payload["ai_interpretation"] = self.ollama_interpretation(payload)

            r = requests.post(
                APPS_SCRIPT_URL,
                json=payload,
                timeout=15,
                allow_redirects=True
            )

            response_text = r.text.strip()

            if r.status_code == 200 and response_text == "OK":
                self.sheet_count += 1
                self.sheet_status = "OK - data + interpretasi 9 aspek terkirim"
            else:
                self.sheet_status = f"ERROR: {response_text[:120]}"

        except Exception as e:
            self.sheet_status = f"ERROR: {str(e)[:120]}"

        finally:
            self.sheet_busy = False
            self.root.after(0, self.refresh_ui)

    def set_local_xyz(self):
        if not self.gps_valid:
            return
        if self.lat == 0.0 and self.lng == 0.0:
            return

        if self.origin_lat is None or self.origin_lng is None:
            self.origin_lat = self.lat
            self.origin_lng = self.lng
            self.origin_alt = self.altitude if self.alt_valid else 0.0

        mean_lat_rad = math.radians((self.lat + self.origin_lat) / 2)
        meter_per_deg_lat = 111320.0
        meter_per_deg_lng = 111320.0 * math.cos(mean_lat_rad)

        self.local_x = (self.lng - self.origin_lng) * meter_per_deg_lng
        self.local_y = (self.lat - self.origin_lat) * meter_per_deg_lat
        self.local_z = (self.altitude - self.origin_alt) if self.alt_valid else 0.0

    def add_track_point(self):
        if not self.gps_valid:
            return
        if self.lat == 0.0 and self.lng == 0.0:
            return

        self.set_local_xyz()
        p = (round(self.local_x, 3), round(self.local_y, 3), round(self.local_z, 3))

        if not self.track or self.track[-1] != p:
            self.track.append(p)

        if len(self.track) > 250:
            self.track.pop(0)

    def draw_track(self):
        c = self.map_canvas
        c.delete("all")

        w = max(c.winfo_width(), 420)
        h = max(c.winfo_height(), 230)

        for x in range(0, w, 35):
            c.create_line(x, 0, x, h, fill="#c8d9e6")
        for y in range(0, h, 35):
            c.create_line(0, y, w, y, fill="#c8d9e6")

        c.create_text(12, 10, anchor="nw", text="GPS Track Preview dalam koordinat lokal", fill="#17324d", font=("Arial", 10, "bold"))
        c.create_text(
            12, 30,
            anchor="nw",
            text=f"X: {self.local_x:.2f} m   Y: {self.local_y:.2f} m   Z: {self.local_z:.2f} m",
            fill="#0f172a",
            font=("Consolas", 9, "bold")
        )
        c.create_text(
            12, 50,
            anchor="nw",
            text="X = timur/barat, Y = utara/selatan, Z = tinggi relatif dari titik awal",
            fill="#475569",
            font=("Arial", 8)
        )

        if len(self.track) < 2:
            c.create_text(w / 2, h / 2, text="Track X/Y/Z muncul setelah GPS valid dan posisi berubah", fill="#555")
            return

        xs = [p[0] for p in self.track]
        ys = [p[1] for p in self.track]

        min_x, max_x = min(xs), max(xs)
        min_y, max_y = min(ys), max(ys)

        if min_x == max_x:
            min_x -= 1
            max_x += 1
        if min_y == max_y:
            min_y -= 1
            max_y += 1

        margin = 42
        plot_top = 72
        plot_bottom = h - margin
        plot_left = margin
        plot_right = w - margin
        pts = []

        def map_x(value):
            return plot_left + (value - min_x) / (max_x - min_x) * (plot_right - plot_left)

        def map_y(value):
            return plot_bottom - (value - min_y) / (max_y - min_y) * (plot_bottom - plot_top)

        # Sumbu lokal X dan Y.
        if min_x <= 0 <= max_x:
            ox = map_x(0)
            c.create_line(ox, plot_top, ox, plot_bottom, fill="#94a3b8", dash=(4, 3))
            c.create_text(ox + 5, plot_top + 5, anchor="nw", text="Y", fill="#64748b", font=("Arial", 8, "bold"))
        if min_y <= 0 <= max_y:
            oy = map_y(0)
            c.create_line(plot_left, oy, plot_right, oy, fill="#94a3b8", dash=(4, 3))
            c.create_text(plot_right - 14, oy - 16, anchor="nw", text="X", fill="#64748b", font=("Arial", 8, "bold"))

        for x_val, y_val, z_val in self.track:
            x = map_x(x_val)
            y = map_y(y_val)
            pts.append((x, y, z_val))

        for i in range(len(pts) - 1):
            c.create_line(pts[i][0], pts[i][1], pts[i + 1][0], pts[i + 1][1], fill="#cc6600", width=3)

        for i, (x, y, z_val) in enumerate(pts):
            r = 7 if i == len(pts) - 1 else 4
            fill = "#d32f2f" if i == len(pts) - 1 else "#1565c0"
            c.create_oval(x - r, y - r, x + r, y + r, fill=fill, outline="")
            if i == len(pts) - 1:
                c.create_text(x + 10, y - 12, anchor="nw", text=f"Z {z_val:.2f} m", fill="#b91c1c", font=("Consolas", 8, "bold"))

        c.create_text(plot_left, h - 22, anchor="nw", text=f"Rentang X: {min_x:.1f} sampai {max_x:.1f} m", fill="#334155", font=("Arial", 8))
        c.create_text(plot_left + 180, h - 22, anchor="nw", text=f"Rentang Y: {min_y:.1f} sampai {max_y:.1f} m", fill="#334155", font=("Arial", 8))

    def draw_graph(self, c, series_list, legends, title):
        c.delete("all")

        w = max(c.winfo_width(), 500)
        h = max(c.winfo_height(), 280)

        left, right, top, bottom = 68, 24, 42, 44

        clean_series = []
        all_values = []
        for s in series_list:
            values = [float(v) for v in s if v is not None]
            clean_series.append(values)
            all_values.extend(values)

        c.create_line(left, top, left, h - bottom, fill="#444", width=2)
        c.create_line(left, h - bottom, w - right, h - bottom, fill="#444", width=2)
        c.create_text(12, 12, anchor="nw", text=title, fill="#111", font=("Arial", 11, "bold"))

        if not all_values:
            c.create_text(w / 2, h / 2, text="Menunggu data real-time", fill="#64748b", font=("Arial", 10, "bold"))
            return

        y_min, y_max = min(all_values), max(all_values)
        if y_min == y_max:
            padding = max(abs(y_max) * 0.05, 5)
            y_min -= padding
            y_max += padding
        else:
            span = y_max - y_min
            y_min -= span * 0.12
            y_max += span * 0.12

        current = all_values[-1]
        c.create_text(
            w - 20, 12,
            anchor="ne",
            text=f"Now {current:.2f} | Min {min(all_values):.2f} | Max {max(all_values):.2f}",
            fill="#334155",
            font=("Consolas", 9, "bold")
        )

        for i in range(6):
            val = y_min + i * (y_max - y_min) / 5
            y = self.y_map(val, h, top, bottom, y_min, y_max)
            c.create_line(left - 5, y, w - right, y, fill="#e5e5e5")
            c.create_text(left - 10, y, text=f"{val:.2f}", anchor="e", fill="#333", font=("Arial", 8))

        colors = ["#1976d2", "#d32f2f", "#388e3c", "#f57c00"]

        for idx, s in enumerate(clean_series):
            if len(s) < 2:
                continue

            step = (w - left - right) / (len(s) - 1)
            coords = []

            for i, val in enumerate(s):
                x = left + i * step
                y = self.y_map(val, h, top, bottom, y_min, y_max)
                coords.extend([x, y])

            c.create_line(*coords, fill=colors[idx % len(colors)], width=3, smooth=True)
            c.create_text(w - 120, 32 + idx * 17, text=legends[idx], fill=colors[idx % len(colors)], font=("Arial", 9, "bold"))


    def y_map(self, value, h, top, bottom, y_min, y_max):
        if y_max == y_min:
            return h - bottom

        plot_top = top
        plot_bottom = h - bottom
        ratio = (value - y_min) / (y_max - y_min)
        y = plot_bottom - ratio * (plot_bottom - plot_top)
        return max(plot_top + 5, min(plot_bottom - 5, y))

    def clear_all(self):

        self.table_rows.clear()

            d.clear()

        self.draw_track()
        self.draw_all_graphs()
